       /// Generation parameters resolved from the preset hierarchy (D-03).
       /// Passed to AiService::stream_chat to configure the AI request.
       #[derive(Clone, Debug, Default)]
       pub struct GenerationParams {
           pub temperature: Option<f64>,
           pub top_p: Option<f64>,
           pub max_tokens: Option<u64>,
           pub frequency_penalty: Option<f64>,
           pub presence_penalty: Option<f64>,
           pub stop_sequences: Option<Vec<String>>,
       }

       impl GenerationParams {
           /// Create from a resolved GenerationPreset (D-01, D-13).
           /// Returns default (all None) if no preset is assigned.
           pub fn from_preset(preset: Option<&GenerationPreset>) -> Self {
               preset.map(|p| Self {
                   temperature: Some(p.temperature),
                   top_p: Some(p.top_p),
                   max_tokens: if p.max_tokens > 0 { Some(p.max_tokens as u64) } else { None },
                   frequency_penalty: if p.frequency_penalty != 0.0 { Some(p.frequency_penalty) } else { None },
                   presence_penalty: if p.presence_penalty != 0.0 { Some(p.presence_penalty) } else { None },
                   stop_sequences: if p.stop_sequences.is_empty() { None } else { Some(p.stop_sequences.clone()) },
               })
               .unwrap_or_default()
           }
       }
       ```

         fn resolve_preset_for_conversation(&self, conv: &Conversation) -> Option<GenerationPreset> {
             // 1. conversation override
             if let Some(ref preset_id) = conv.generation_preset_id {
                 if let Some(preset) = self.presets.iter().find(|p| p.id == *preset_id) {
                     return Some(preset.clone());
                 }
             }
             // 2. character default
             if let Some(char) = self.characters.iter().find(|c| c.id == conv.character_id) {
                 if let Some(ref preset_id) = char.default_preset_id {
                     if let Some(preset) = self.presets.iter().find(|p| p.id == *preset_id) {
                         return Some(preset.clone());
                     }
                 }
             }
             // 3. global default
             if let Some(ref preset_id) = self.global_default_preset_id {
                 if let Some(preset) = self.presets.iter().find(|p| p.id == *preset_id) {
                     return Some(preset.clone());
                 }
             }
             // 4. no preset → provider defaults (D-13)
             None
         }
         ```
