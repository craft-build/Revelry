/// Standalone lorebook — a named collection of keyword-activated entries.
#[derive(Clone, Debug, Serialize, Deserialize, PartialEq)]
pub struct Lorebook {
    pub id: String,

            description: String::new(),
            created_at: now,
            updated_at: now,
        }
    }
}

/// LorebookPosition: where the entry injects relative to character definition.
#[derive(Clone, Debug, Serialize, Deserialize, PartialEq)]
pub enum LorebookPosition {
    BeforeChar,
    AfterChar,
    AtDepth,
}

impl std::fmt::Display for LorebookPosition {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            LorebookPosition::BeforeChar => write!(f, "before_char"),
            LorebookPosition::AfterChar => write!(f, "after_char"),
            LorebookPosition::AtDepth => write!(f, "at_depth"),
        }
    }
}

impl std::str::FromStr for LorebookPosition {
    type Err = String;
    fn from_str(s: &str) -> Result<Self, Self::Err> {
        match s.to_lowercase().as_str() {
            "before_char" => Ok(LorebookPosition::BeforeChar),
            "after_char" => Ok(LorebookPosition::AfterChar),

use crate::domain::lorebook::{Lorebook, LorebookEntry, LorebookPosition, LorebookActivation};
use crate::db::models;
use rusqlite::Connection;

pub fn load_lorebooks(conn: &Connection) -> anyhow::Result<Vec<Lorebook>> {
    let mut stmt = conn.prepare(
        "SELECT id, name, description, created_at, updated_at FROM lorebooks ORDER BY name"
    )?;
    let lorebooks = stmt.query_map([], |row| models::row_to_lorebook(row))?;
    lorebooks.collect::<Result<Vec<_>, _>>().map_err(|e| anyhow::anyhow!("Failed to load lorebooks: {}", e))
}

pub fn load_lorebook(conn: &Connection, id: &str) -> anyhow::Result<Option<Lorebook>> {
    let mut stmt = conn.prepare(
        "SELECT id, name, description, created_at, updated_at FROM lorebooks WHERE id = ?1"
    )?;
    let result = stmt.query_row([id], |row| models::row_to_lorebook(row));
    match result {
        Ok(lorebook) => Ok(Some(lorebook)),
        Err(rusqlite::Error::QueryReturnedNoRows) => Ok(None),
        Err(e) => Err(anyhow::anyhow!("Failed to load lorebook: {}", e)),
    }
}

pub fn insert_lorebook(conn: &Connection, lorebook: &Lorebook) -> anyhow::Result<()> {
    conn.execute(
        "INSERT INTO lorebooks (id, name, description, created_at, updated_at) VALUES (?1, ?2, ?3, ?4, ?5)",
        [&lorebook.id, &lorebook.name, &lorebook.description, &lorebook.created_at.to_rfc3339(), &lorebook.updated_at.to_rfc3339()],
    ).map_err(|e| anyhow::anyhow!("Failed to insert lorebook: {}", e))?;
    Ok(())
}

pub fn update_lorebook(conn: &Connection, lorebook: &Lorebook) -> anyhow::Result<()> {
    conn.execute(
        "UPDATE lorebooks SET name = ?2, description = ?3, updated_at = ?4 WHERE id = ?1",
        [&lorebook.id, &lorebook.name, &lorebook.description, &lorebook.updated_at.to_rfc3339()],
    ).map_err(|e| anyhow::anyhow!("Failed to update lorebook: {}", e))?;
    Ok(())
}

pub fn delete_lorebook(conn: &Connection, id: &str) -> anyhow::Result<()> {

    Ok(())
}

    entries.collect::<Result<Vec<_>, _>>().map_err(|e| anyhow::anyhow!("Failed to load lorebook entries: {}", e))
