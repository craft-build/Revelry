     ```rust
     use relm4::prelude::*;
     
     pub struct DbService {
         conn: Connection,  // !Send — stays on worker thread
     }
     
     impl Worker for DbService {
         type Init = PathBuf;  // Database file path
         type Input = DbInput;
         type Output = DbOutput;
         
         fn init(db_path: Self::Init, sender: ComponentSender<Self>) -> Self {
             let conn = Connection::open(&db_path).expect("Failed to open database");
             run_migrations(&conn).expect("Failed to run migrations");
             DbService { conn }
         }
         
         fn update(&mut self, msg: DbInput, sender: ComponentSender<Self>) {
             let result = match msg {
                 DbInput::LoadProviders => {
                     providers::load_providers(&self.conn)
                         .map(DbOutput::ProvidersLoaded)
                 }
                 DbInput::SaveProvider(p) => {
                     providers::insert_provider(&self.conn, &p)
                         .map(|_| DbOutput::ProviderSaved(p))
                 }
                 DbInput::DeleteProvider(id) => {
                     providers::delete_provider(&self.conn, &id)
                         .map(|_| DbOutput::ProviderDeleted(id))
                 }
                 DbInput::LoadAppSetting(key) => {
                     // ... implementation
                 }
                 DbInput::SaveAppSetting { key, value } => {
                     // ... implementation
                 }
             };
             match result {
                 Ok(output) => sender.output(output).unwrap(),
                 Err(e) => sender.output(DbOutput::Error(e.to_string())).unwrap(),
             }
         }
     }
     ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Application ↔ SQLite | Provider metadata stored in local database file. File permissions must be restrictive (0600 on Unix). |
| Application ↔ OS Keychain | API keys stored in OS credential store. Keychain may be unavailable (headless Linux, locked keychain). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Information Disclosure | providers table | mitigate | No api_key column in schema — keys stored exclusively in OS keychain (D-16, PROV-05) |
| T-01-02 | Denial of Service | DbService Worker | accept | SQLite busy_timeout handles lock contention. Single-worker serializes access. Low risk for local app. |
| T-01-03 | Tampering | SQLite database file | accept | Local-first app, single user. File permissions set by OS. Vault encryption deferred to Phase 6. |
</threat_model>
