2. Add two new async methods to the `AppState` class, after the `uploadAvatar` method:

```typescript
  // ── Card import/export ────────────────────────────────────────────────

  /** Open a file picker for card import, parse the file via backend,
   *  and return the character data for the editor to consume. */
  async importCharacterFromFile(): Promise<CharacterCard> {
    const filePath = await open({
      multiple: false,
      filters: [
        { name: "Character Cards", extensions: ["png", "json"] },
      ],
    });

    if (!filePath) {
      throw new Error("cancelled");
    }

    // open() returns string | string[] | null. With multiple: false, it's string | null.
    const path = typeof filePath === "string" ? filePath : filePath;

    return await apiImportCard(path);
  }

  /** Export a character card as a TavernAI V2 PNG file.
   *  Opens a native save dialog and writes the file via backend. */
  async exportCharacter(id: string): Promise<void> {
    const character = this.characters.find((c) => c.id === id);
    if (!character) throw new Error("Character not found");

    const savePath = await save({
      defaultPath: `${character.name}.png`,
      filters: [
        { name: "PNG Image", extensions: ["png"] },
      ],
    });

    if (!savePath) return; // User cancelled

    await apiExportCard(id, savePath);
  }
```

```typescript
async function handleImport() {
  if (is_importing) return;
  is_importing = true;
  try {
    const importedCharacter = await app.importCharacterFromFile();
    // Open the character editor in "create" mode, pre-filled with imported data
    editor_mode = "create";
    editing_character = importedCharacter;
    editor_open = true;
  } catch (err) {
    // User cancelled file dialog — not an error
    if (err instanceof Error && err.message === "cancelled") {
      return;
    }
    // Show import error dialog
    import_error_message =
      err instanceof Error
        ? err.message
        : typeof err === "string"
          ? err
          : "Could not read the character card.";
    import_error_open = true;
  } finally {
    is_importing = false;
  }
}
```
