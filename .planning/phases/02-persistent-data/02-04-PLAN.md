  return invoke("delete_character", { id });

  query: string,

  return invoke<CharacterCard[]>("search_characters", { query });

export async function getPersonas(): Promise<UserPersona[]> {
  return invoke<UserPersona[]>("get_personas");

  return invoke<UserPersona>("save_persona", { persona });
}

export async function deletePersona(id: string): Promise<void> {

export async function searchPersonas(query: string): Promise<UserPersona[]> {

  return invoke("delete_conversation", { id });

export async function searchConversations(
  query: string,

  return invoke<Conversation[]>("search_conversations", { query });
}

// ── Messages ─────────────────────────────────────────────────────────
export async function getMessages(conversationId: string): Promise<Message[]> {
  return invoke<Message[]>("get_messages", { conversationId });

  return invoke("save_message", { message });

Key details:

- **Tauri v2 command naming (RESOLVED):** Tauri v2 uses the Rust function name as-is (snake_case) for the command name. The `invoke()` call MUST use the exact same snake_case string (e.g., `'get_characters'`, `'search_personas'`, `'update_conversation_persona'`). Tauri v2 does NOT auto-convert to camelCase.
- For command arguments, Tauri v2 DOES apply camelCase-to-snake_case conversion. So the frontend sends `{ conversationId }` which maps to the Rust `conversation_id` parameter, and `{ personaId }` maps to `persona_id`.
