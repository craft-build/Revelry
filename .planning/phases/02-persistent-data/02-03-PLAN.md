```rust
use tauri::State;
use crate::db::{DbCommand, DbWorker};
use crate::domain::CharacterCardV2;

#[tauri::command]
pub async fn get_characters(db: State<'_, DbWorker>) -> Result<Vec<CharacterCardV2>, String> {
    let (reply_tx, reply_rx) = tokio::sync::oneshot::channel();
    db.send(DbCommand::LoadCharacters { reply: reply_tx })
        .map_err(|e| e.to_string())?;
    reply_rx.await
        .map_err(|e| format!("DB worker dropped: {}", e))?
        .map_err(|e| e.to_string())
}

#[tauri::command]
pub async fn save_character(db: State<'_, DbWorker>, character: CharacterCardV2) -> Result<CharacterCardV2, String> {
    let (reply_tx, reply_rx) = tokio::sync::oneshot::channel();
    db.send(DbCommand::SaveCharacter { character, reply: reply_tx })
        .map_err(|e| e.to_string())?;
    reply_rx.await
        .map_err(|e| format!("DB worker dropped: {}", e))?
        .map_err(|e| e.to_string())
}

#[tauri::command]
pub async fn delete_character(db: State<'_, DbWorker>, id: String) -> Result<(), String> {
    let (reply_tx, reply_rx) = tokio::sync::oneshot::channel();
    db.send(DbCommand::DeleteCharacter { id, reply: reply_tx })
        .map_err(|e| e.to_string())?;
    reply_rx.await
        .map_err(|e| format!("DB worker dropped: {}", e))?
        .map_err(|e| e.to_string())
}

#[tauri::command]
pub async fn search_characters(db: State<'_, DbWorker>, query: String) -> Result<Vec<CharacterCardV2>, String> {
    let (reply_tx, reply_rx) = tokio::sync::oneshot::channel();
    db.send(DbCommand::SearchCharacters { query, reply: reply_tx })
        .map_err(|e| e.to_string())?;
    reply_rx.await
        .map_err(|e| format!("DB worker dropped: {}", e))?
        .map_err(|e| e.to_string())
}
```

Note: `delete_character` returns `Result<(), String>` (not `Result<String, String>`) — the DbCommand variant carries `oneshot::Sender<anyhow::Result<()>>`.

**Tauri v2 command naming convention (RESOLVED):** Tauri v2 uses the Rust function name as-is (snake_case) for the command name. The frontend `invoke()` MUST use the exact same snake_case string (e.g., `'get_characters'`, `'search_personas'`, `'update_conversation_persona'`). Tauri v2 does NOT auto-convert to camelCase. For command arguments, Tauri v2 DOES apply camelCase-to-snake_case conversion, so the frontend sends `{ characterId }` which maps to the Rust `character_id` parameter.

```rust
mod db;
mod domain;
mod commands;

use db::DbWorker;

#[cfg_attr(mobile, tauri::mobile_entry_point)]
pub fn run() {
    // Resolve database path in platform-appropriate data directory
    let db_path = dirs::data_dir()
        .expect("Could not determine platform data directory")
        .join("com.revelry.desktop")
        .join("revelry.db");

    // Start DB worker (creates directory + opens connection + runs migrations + seeds presets)
    let db_worker = DbWorker::start(db_path.clone())
        .unwrap_or_else(|e| {
            eprintln!("FATAL: Failed to start database worker at {:?}: {}", db_path, e);
            std::process::exit(1);
        });

    tauri::Builder::default()
        .plugin(tauri_plugin_fs::init())
        .plugin(tauri_plugin_dialog::init())
        .plugin(tauri_plugin_shell::init())
        .plugin(tauri_plugin_http::init())
        .plugin(tauri_plugin_store::Builder::default().build())
        .manage(db_worker)
        .invoke_handler(tauri::generate_handler![
            commands::characters::get_characters,
            commands::characters::save_character,
            commands::characters::delete_character,
            commands::characters::search_characters,
            commands::personas::get_personas,
            commands::personas::save_persona,
            commands::personas::delete_persona,
            commands::conversations::get_conversations,
            commands::conversations::start_conversation,
            commands::conversations::delete_conversation,
            commands::conversations::search_conversations,
            commands::conversations::get_messages,
            commands::conversations::save_message,
            commands::conversations::update_conversation_persona,
            commands::personas::search_personas,
            commands::presets::get_presets,
        ])
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}
```

Note: The `unwrap_or_else` with `std::process::exit(1)` ensures the app fails loudly if DB init fails, rather than silently continuing with a broken state. The eprintln message includes the DB path for user diagnostics. This is consistent with the UI-SPEC's "Database Initialization Failure" error state — the Rust side exits, and the frontend will see the app crash, which will be improved in plan 02-04 with a proper error dialog.
